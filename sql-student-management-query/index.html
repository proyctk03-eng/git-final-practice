Yêu cầu Tiêu chí 1 (+20đ): Hiển thị tất cả các sinh viên có tên bắt đầu bằng ký tự 'h'.
Phương pháp: Sử dụng mệnh đề WHERE StudentName LIKE 'h%'. Trong MySQL collation utf8mb4, phép so khớp không phân biệt hoa thường nên nhận diện cả 'Hung' và 'Hoa'.
MySQL Query - Tiêu chí 1 
SELECT * 
FROM Student 
WHERE StudentName LIKE 'h%';

1 Hung Ha Noi 0912113113 1 (Đang học) 1
2 Hoa Hai phong NULL 1 (Đang học) 1
Yêu cầu Tiêu chí 2 (+20đ): Hiển thị các thông tin lớp học có thời gian bắt đầu vào tháng 12.
Phương pháp: Sử dụng hàm xử lý thời gian MONTH(StartDate) = 12 để trích xuất chỉ số tháng từ cột kiểu DATETIME.

2 A2 2008-12-22 00:00:00 1 (Hoạt động)
Yêu cầu Tiêu chí 3 (+20đ): Hiển thị tất cả các thông tin môn học có credit trong khoảng từ 3-5.
Phương pháp: Sử dụng toán tử BETWEEN 3 AND 5 để xác định chính xác đoạn giá trị đóng [3, 5], loại bỏ môn học có số tín chỉ ngoài khoảng (như C: 6 tín chỉ, RDBMS: 10 tín chỉ).

Yêu cầu Tiêu chí 4 (+20đ): Thay đổi mã lớp (ClassID) của sinh viên có tên 'Hung' là 2.
Phương pháp: Sử dụng câu lệnh UPDATE Student SET ClassId = 2 WHERE StudentName = 'Hung'; và chạy câu lệnh SELECT để xác minh kết quả cập nhật.

Yêu cầu Tiêu chí 5 (+20đ): Hiển thị các thông tin: StudentName, SubName, Mark. Dữ liệu sắp xếp theo điểm thi (mark) giảm dần, nếu trùng sắp theo tên tăng dần.
Phương pháp: JOIN 3 bảng (Student, Mark, Subject) và áp dụng mệnh đề ORDER BY M.Mark DESC, S.StudentName ASC.
MySQL Query - Tiêu chí 5 
SELECT 
    S.StudentName, 
    Sub.SubName, 
    M.Mark 
FROM Student S 
JOIN Mark M ON S.StudentId = M.StudentId 
JOIN Subject Sub ON M.SubId = Sub.SubId 
ORDER BY M.Mark DESC, S.StudentName ASC;

-- 1. KHOI TAO CSDL VA DU LIEU MAU
CREATE DATABASE IF NOT EXISTS QuanLySinhVien;
USE QuanLySinhVien;

CREATE TABLE IF NOT EXISTS Class (
    ClassID INT NOT NULL AUTO_INCREMENT PRIMARY KEY,
    ClassName VARCHAR(60) NOT NULL,
    StartDate DATETIME NOT NULL,
    Status BIT
);

CREATE TABLE IF NOT EXISTS Student (
    StudentId INT NOT NULL AUTO_INCREMENT PRIMARY KEY,
    StudentName VARCHAR(30) NOT NULL,
    Address VARCHAR(50),
    Phone VARCHAR(20),
    Status BIT,
    ClassId INT NOT NULL,
    CONSTRAINT FK_Student_Class FOREIGN KEY (ClassId) REFERENCES Class (ClassID)
);

CREATE TABLE IF NOT EXISTS Subject (
    SubId INT NOT NULL AUTO_INCREMENT PRIMARY KEY,
    SubName VARCHAR(30) NOT NULL,
    Credit TINYINT NOT NULL DEFAULT 1 CHECK (Credit >= 1),
    Status BIT DEFAULT 1
);

CREATE TABLE IF NOT EXISTS Mark (
    MarkId INT NOT NULL AUTO_INCREMENT PRIMARY KEY,
    SubId INT NOT NULL,
    StudentId INT NOT NULL,
    Mark FLOAT DEFAULT 0 CHECK (Mark BETWEEN 0 AND 100),
    ExamTimes TINYINT DEFAULT 1,
    UNIQUE (SubId, StudentId),
    CONSTRAINT FK_Mark_Subject FOREIGN KEY (SubId) REFERENCES Subject (SubId),
    CONSTRAINT FK_Mark_Student FOREIGN KEY (StudentId) REFERENCES Student (StudentId)
);

-- 2. 5 CAU TRUY VAN CHINH THEO TIÊU CHÍ CHẤM ĐIỂM (100/100 DIEM)
USE QuanLySinhVien;

-- Tieu chi 1: Ten bat dau bang 'h'
SELECT * FROM Student WHERE StudentName LIKE 'h%';

-- Tieu chi 2: Thoi gian bat dau vao thang 12
SELECT * FROM Class WHERE MONTH(StartDate) = 12;

-- Tieu chi 3: Credit trong khoang 3-5
SELECT * FROM Subject WHERE Credit BETWEEN 3 AND 5;

-- Tieu chi 4: Thay doi ma lop cua sinh vien 'Hung' la 2
SET SQL_SAFE_UPDATES = 0;
UPDATE Student SET ClassId = 2 WHERE StudentName = 'Hung';
SET SQL_SAFE_UPDATES = 1;
SELECT * FROM Student WHERE StudentName = 'Hung';

-- Tieu chi 5: ORDER BY Mark giam dan, StudentName tang dan
SELECT S.StudentName, Sub.SubName, M.Mark 
FROM Student S 
JOIN Mark M ON S.StudentId = M.StudentId 
JOIN Subject Sub ON M.SubId = Sub.SubId 
ORDER BY M.Mark DESC, S.StudentName ASC;
