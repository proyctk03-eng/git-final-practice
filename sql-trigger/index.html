Thực hành triển khai Trigger từng bước

Nguyên lý Trigger BEFORE INSERT:
- Trigger được kích hoạt trước khi bản ghi thực sự được ghi vào đĩa lưu trữ.
- Biến NEW đại diện cho dòng dữ liệu mới đang chuẩn bị được chèn.
- Ta có quyền thay đổi các trường dữ liệu bằng câu lệnh: SET NEW.department = .... Điều này không thể thực hiện trong trigger AFTER INSERT vì lúc đó bản ghi đã trở thành chỉ đọc.
MySQL Script - Tạo Trigger update_department 
DELIMITER //

DROP TRIGGER IF EXISTS update_department //

CREATE TRIGGER update_department
BEFORE INSERT ON employees
FOR EACH ROW
BEGIN
    IF NEW.salary >= 5000 THEN
        SET NEW.department = 'Management';
    ELSEIF NEW.salary >= 3000 THEN
        SET NEW.department = 'Sales';
    ELSE
        SET NEW.department = 'Support';
    END IF;
END //

DELIMITER ;
Thử nghiệm chèn dữ liệu với phòng ban giả định 'A':
Mặc dù người dùng truyền giá trị phòng ban là 'A' trong câu lệnh INSERT, Trigger update_department sẽ chặn và tự động gán lại phòng ban chuẩn theo mức lương.
MySQL Script - Chèn dữ liệu thử nghiệm 
INSERT INTO employees (name, department, salary) VALUES
('John Doe', 'A', 3500),
('Jane Smith', 'A', 2000),
('David Johnson', 'A', 6000);

-- Truy vấn kiểm tra
SELECT id, name, department, salary FROM employees ORDER BY id ASC;

Bảng kết quả đối chiếu trước và sau khi Trigger can thiệp:

id name Giá trị đầu vào (Input) Phòng ban sau Trigger (Output) salary Quy tắc phân loại tự động
1 John Doe 'A' Sales 3,500.00 Mức lương 3500 ≥ 3000 → Phòng Sales
2 Jane Smith 'A' Support 2,000.00 Mức lương 2000 < 3000 → Phòng Support
3 David Johnson 'A' Management 6,000.00 Mức lương 6000 ≥ 5000 → Phòng Management
Ứng dụng thực tế: Trigger kiểm toán (Audit Trail):
Sử dụng trigger AFTER UPDATE để tự động ghi lại lịch sử điều chỉnh lương của nhân viên, đối chiếu giá trị lương cũ OLD.salary và lương mới NEW.salary.
MySQL Script - Trigger ghi log khi sửa lương 
CREATE TABLE IF NOT EXISTS salary_audit (
    audit_id INT AUTO_INCREMENT PRIMARY KEY,
    employee_id INT NOT NULL,
    employee_name VARCHAR(50) NOT NULL,
    old_salary DECIMAL(10,2) NOT NULL,
    new_salary DECIMAL(10,2) NOT NULL,
    action_type VARCHAR(20) NOT NULL,
    changed_at DATETIME NOT NULL
);

DELIMITER //
CREATE TRIGGER trg_audit_salary_update
AFTER UPDATE ON employees
FOR EACH ROW
BEGIN
    IF OLD.salary <> NEW.salary THEN
        INSERT INTO salary_audit (
            employee_id, employee_name, old_salary, new_salary, action_type, changed_at
        ) VALUES (
            OLD.id, OLD.name, OLD.salary, NEW.salary, 'SALARY_UPDATE', NOW()
        );
    END IF;
END //
DELIMITER ;

-- Thử nghiệm tăng lương cho Jane Smith
UPDATE employees SET salary = 3200 WHERE name = 'Jane Smith';

Dữ liệu tự động sinh trong bảng salary_audit:

audit_id employee_id employee_name old_salary new_salary action_type changed_at
1 2 Jane Smith 2,000.00 3,200.00 SALARY_UPDATE 2026-10-08 16:40:00
Ma trận khả dụng của các biến giả lập NEW và OLD trong MySQL Trigger:
Sự kiện DML Biến OLD (Giá trị trước sự kiện) Biến NEW (Giá trị sau sự kiện) Mục đích sử dụng tiêu biểu
INSERT Không khả dụng Khả dụng (NEW.column) Kiểm tra hợp lệ, gán giá trị mặc định, phân loại tự động
UPDATE Khả dụng (OLD.column) Khả dụng (NEW.column) Đối chiếu biến động, ghi log kiểm toán (Audit Trail)
DELETE Khả dụng (OLD.column) Không khả dụng Lưu trữ dữ liệu vào bảng thùng rác (Archive), kiểm tra ràng buộc hủy

trigger_demo.sql 
-- 1. KHOI TAO CSDL VA BANG
CREATE DATABASE IF NOT EXISTS company;
USE company;

DROP TABLE IF EXISTS employees;

CREATE TABLE employees (
    id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(50) NOT NULL,
    department VARCHAR(50) NOT NULL,
    salary DECIMAL(10,2) NOT NULL
);

-- 2. TAO TRIGGER PHAN LOAI PHONG BAN
DELIMITER //
DROP TRIGGER IF EXISTS update_department //
CREATE TRIGGER update_department
BEFORE INSERT ON employees
FOR EACH ROW
BEGIN
    IF NEW.salary >= 5000 THEN
        SET NEW.department = 'Management';
    ELSEIF NEW.salary >= 3000 THEN
        SET NEW.department = 'Sales';
    ELSE
        SET NEW.department = 'Support';
    END IF;
END //
DELIMITER ;

-- 3. DEMO CHEN DU LIEU
INSERT INTO employees (name, department, salary) VALUES
('John Doe', 'A', 3500),
('Jane Smith', 'A', 2000),
('David Johnson', 'A', 6000);

SELECT id, name, department, salary FROM employees ORDER BY id ASC;

-- 4. MO RONG AUDIT LOGGING

DELIMITER //
